import { ReactNode } from 'react';

interface ButtonProps {
  children: ReactNode;
  variant?: 'primary' | 'secondary';
  onClick?: () => void;
}

export const Button = ({ children, variant = 'secondary', onClick }: ButtonProps) => {
  const baseStyle = "px-3 py-1.5 rounded-md text-sm font-medium transition-colors border";
  const variants = {
    primary: "bg-primary text-black border-primary hover:opacity-90 shadow-[0_0_15px_rgba(147,61,245,0.3)]",
    secondary: "bg-transparent text-text-primary border-border-dark hover:border-text-muted"
  };

  return (
    <button className={{`${baseStyle ${variants[variant]} onClick={onClick}>
      {children}
    </button>
  );
};
